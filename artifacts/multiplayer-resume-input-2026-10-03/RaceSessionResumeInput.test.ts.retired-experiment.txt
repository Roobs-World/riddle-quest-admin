import {expect, it} from "@jest/globals";
import {racing} from "./raceSession.support";

it.each([false, true])("keeps mutations blocked while composing a recovery draft (committed=%s)", async committed => {
  /** Authority may have accepted the failed wrong guess or never received it. */
  const h = await racing(); h.session.input("text", "zzzzz"); h.session.submit(); h.session.input("text", "pi");
  if (committed) await h.authority.submitRaceGuess({runId: h.run.runId, riddleId: h.run.currentRiddle!.riddleId,
    guess: "zzzzz", requestId: "committed-wrong"});
  h.reject("raceGuess", "Lost confirmation"); h.session.retry();
  expect(h.session.getSnapshot()).toMatchObject({allowTypingWhilePending: true, currentLetters: ["p", "i", "", "", ""], requiresRunRefresh: true});
  /** Deliberate slot and IME text edits change only the local draft. */
  h.session.input("slot", 3); h.session.input("text", "no");
  const count = h.commands.length; h.session.submit(); h.session.hint(); h.session.skip();
  expect(h.commands).toHaveLength(count);
  h.respond("raceResume", await h.authority.resumeRace(h.run.runId));
  expect(h.session.getSnapshot()).toMatchObject({currentLetters: ["p", "i", "", "n", "o"], activeSlotIndex: 2,
    pending: null, requiresRunRefresh: false, run: {currentRiddle: {guessesUsed: committed ? 1 : 0}}});
});

it("retains a deliberately emptied draft through repeated failed recovery reads", async () => {
  /** No submission projection exists to own edits made during a plain reconnect read. */
  const h = await racing(); h.session.input("text", "pi"); h.session.retry();
  h.session.input("backspace"); h.session.input("backspace");
  h.reject("raceResume", "Read failed"); h.session.retry();
  h.respond("raceResume", await h.authority.resumeRace(h.run.runId));
  expect(h.session.getSnapshot().currentLetters).toEqual(Array(5).fill(""));
});

it("applies new host hint locks over letters composed during the recovery read", async () => {
  /** The current question remains the same while another tab obtains a hint. */
  const h = await racing(); h.session.input("text", "xi"); h.session.retry(); h.session.input("letter", "a");
  const hinted = await h.authority.requestRaceHint({runId: h.run.runId, riddleId: h.run.currentRiddle!.riddleId, requestId: "other-tab-hint"});
  h.respond("raceResume", {run: hinted.run});
  expect(h.session.getSnapshot()).toMatchObject({currentLetters: ["p", "i", "a", "", ""], run: {totalPenaltyMs: hinted.run.totalPenaltyMs}});
  h.session.input("slot", 0); h.session.input("letter", "z");
  expect(h.session.getSnapshot().currentLetters[0]).toBe("p");
});

it("discards reconnect letters when authority confirms another tab's successor", async () => {
  /** Unsent letters belong to the displayed question rather than a guessed next position. */
  const h = await racing(); h.session.input("text", "pi"); h.session.retry(); h.session.input("letter", "a");
  const advanced = await h.authority.submitRaceGuess({runId: h.run.runId, riddleId: h.run.currentRiddle!.riddleId,
    guess: "piano", requestId: "other-tab-correct"});
  h.respond("raceResume", {run: advanced.run});
  expect(h.session.getSnapshot()).toMatchObject({currentLetters: Array(5).fill(""), run: {currentRiddle: {position: 1}}});
});

it.each([false, true])("keeps projected successor input separate from edits to the reconnect root (advanced=%s)", async advanced => {
  /** A lost correct receipt retains a privately known successor draft before rollback. */
  const h = await racing(); h.session.input("text", "piano"); h.session.submit(); h.session.input("text", "cl");
  h.reject("raceGuess", "Lost confirmation"); h.session.retry();
  /** Root editing must not replace the retained successor's letters. */
  h.session.input("slot", 0); h.session.input("letter", "x");
  const resumed = advanced ? (await h.authority.submitRaceGuess({runId: h.run.runId, riddleId: h.run.currentRiddle!.riddleId,
    guess: "piano", requestId: "committed-correct"})).run : (await h.authority.resumeRace(h.run.runId)).run;
  h.respond("raceResume", {run: resumed});
  expect(h.session.getSnapshot().currentLetters).toEqual(advanced ? ["c", "l", "", "", ""] : ["x", "i", "a", "n", "o"]);
});

it.each(["owner", "locale"])("retires pending reconnect composition after changing %s", async field => {
  /** Old read and letters belong to the prior authenticated journey. */
  const h = await racing(); h.session.input("text", "pi"); h.session.retry(); h.session.input("letter", "a");
  const old = h.last(); h.session.enter({ownerId: field === "owner" ? "owner-2" : "owner-1", locale: field === "locale" ? "fr" : "en"});
  const fresh = h.session.getSnapshot(); h.respond("raceResume", await h.authority.resumeRace(h.run.runId), old);
  expect(h.session.getSnapshot()).toBe(fresh);
  expect(fresh).toMatchObject({run: null, currentLetters: [], allowTypingWhilePending: false});
});

it("retains the latest unsent draft composed while a canonical reconnect read is pending", async () => {
  /** Lost receipt leaves the race requiring an authoritative read before another mutation. */
  const h = await racing();
  h.session.input("text", "zzzzz"); h.session.submit(); h.session.input("text", "pi");
  h.reject("raceGuess", "Lost confirmation"); h.session.retry();
  expect(h.session.getSnapshot().pending).toBe("raceResume");
  /** Composition stays responsive during read latency but never sends another guess. */
  h.session.input("letter", "a");
  expect(h.commands.filter(command => command.type === "game:submitRaceGuess")).toHaveLength(1);
  h.respond("raceResume", await h.authority.resumeRace(h.run.runId));
  expect(h.session.getSnapshot().currentLetters).toEqual(["p", "i", "a", "", ""]);
});

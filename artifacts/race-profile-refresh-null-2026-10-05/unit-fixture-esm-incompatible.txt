import {afterEach, expect, it, jest} from "@jest/globals";
import type {BridgeProfile, HostToGameMessage} from "@qwordly/core/bridge";
import type {AuthSnapshot} from "../types";

jest.mock("react", () => ({useCallback: <T>(callback: T) => callback,
  useMemo: <T>(factory: () => T) => factory(), useRef: <T>(current: T) => ({current}),
  useState: <T>(initial: T) => [initial, jest.fn()], useEffect: () => undefined}));
jest.mock("../../analytics/events", () => ({trackEvent: jest.fn()}));
import {useHostRaceCoordination} from "../useHostRaceCoordination";
import {createRedditHostServices} from "../../reddit/RedditHostServices";

/** Complete gateway profile exercises production schema validation. */
const profile: BridgeProfile = {username: "Player", coins: 300, xp: 0, level: 1, gamesPlayed: 0,
  wins: 0, winStreak: 0, achievements: [], classicIndex: 1};
/** Same-origin gateway envelope for real Reddit profile reads. */
const response = (value: BridgeProfile) => new Response(JSON.stringify({ok: true, result: value}),
  {status: 200, headers: {"content-type": "application/json"}});
/** Original global fetch is restored after the injected gateway finishes. */
const originalFetch = globalThis.fetch;
afterEach(() => {globalThis.fetch = originalFetch;});

it("keeps the subscribed newer profile when Reddit supersedes a race refresh", async () => {
  /** Later gateway reads can remain pending across independently published profile updates. */
  const held: Array<(value: Response) => void> = [];
  /** Startup authority read publishes normally before the race read is held. */
  let hold = false;
  globalThis.fetch = jest.fn<typeof fetch>(async input => {
    expect(String(input)).toBe("/api/game/profile.load");
    return hold ? new Promise<Response>(resolve => held.push(resolve)) : response(profile);
  });
  Object.defineProperty(globalThis, "window", {configurable: true, value: {location: {search: ""}}});
  /** Actual provider revision ordering, not a simulated null-returning loader. */
  const identity = createRedditHostServices();
  await identity.loadProfile("reddit-account");
  /** Subscribed projection observes the actual newest canonical profile. */
  let current: AuthSnapshot = {user: null, profile: null};
  /** Fixture subscription reproduces the host's provider-owned profile publication. */
  const unsubscribe = identity.auth.subscribe(snapshot => {current = snapshot;});
  /** Null must not replace the newer subscribed state. */
  const replaceProfile = jest.fn((value: BridgeProfile | null) => {current = {...current, profile: value};});
  /** Only available profiles may produce the optional explicit race auth reply. */
  const postToGame = jest.fn<(message: HostToGameMessage) => void>();
  /** Real race coordination receives the current provider and account owner. */
  const coordination = useHostRaceCoordination({authSnapshot: current, bridgeUser: current.user, locale: "en",
    race: undefined, identity, raceLocalePolicy: "all", applyCoinBalance: jest.fn(), replaceProfile,
    ownsAccount: uid => current.user?.uid === uid, captureAccount: uid => () => current.user?.uid === uid,
    postToGame});
  hold = true;
  /** Optional post-race profile read is admitted before another caller's newer refresh. */
  const race = coordination.syncProgress({});
  expect(held).toHaveLength(1);
  /** Independent profile refresh supersedes the race read at the actual adapter seam. */
  const newer = identity.loadProfile("reddit-account");
  held[1](response({...profile, coins: 777}));
  expect(await newer).toMatchObject({coins: 777});
  expect(current.profile?.coins).toBe(777);
  held[0](response(profile));
  expect(await race).toBe(true);
  expect(current.profile?.coins).toBe(777);
  expect(replaceProfile).not.toHaveBeenCalled();
  expect(postToGame).not.toHaveBeenCalled();
  unsubscribe();
});

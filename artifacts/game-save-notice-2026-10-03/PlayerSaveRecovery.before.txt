/** Retries failed client saves without continuous polling or work after the owning view retires. */
export class PlayerSaveRecovery {
  /** Increasing delays bound each failure/reconnection episode to three attempts. */
  private readonly delays = [1000, 2000, 4000];
  /** Mount lifetime permits asynchronous retries. */
  private attached = false;
  /** Failure remains recoverable after the automatic attempt budget is exhausted. */
  private pending = false;
  /** Number of automatic attempts in the current episode. */
  private attempts = 0;
  /** Only one future retry may be scheduled. */
  private timer: ReturnType<typeof setTimeout> | null = null;
  /** Retired callbacks cannot reschedule work for a replacement session. */
  private revision = 0;
  /** One running attempt belongs to its own lifetime token. */
  private running: object | null = null;

  /** Executes a retry through the caller's existing durability queue. */
  private readonly retry: () => Promise<void>;
  /** Rechecks browser and target availability before entering an attempt. */
  private readonly available: () => boolean;

  /** Owns retry policy while the caller owns durability and target/account validation. */
  constructor(
    retry: () => Promise<void>,
    available: () => boolean,
  ) {this.retry = retry; this.available = available;}

  /** Enables retry scheduling when the owning view mounts. */
  attach(): void {this.attached = true; this.schedule();}
  /** Retires timers and asynchronous completions when the view unmounts. */
  detach(): void {this.attached = false; this.recovered();}
  /** Marks accepted state as unsaved without replenishing a failing episode's retry budget. */
  failed(): void {this.pending = true; this.schedule();}
  /** Cancels retries only when the caller has acknowledged every latest dirty field. */
  recovered(): void {
    this.pending = false;
    this.attempts = 0;
    this.revision++;
    this.running = null;
    if (this.timer !== null) clearTimeout(this.timer);
    this.timer = null;
  }
  /** A real online/resume signal grants one fresh bounded recovery episode. */
  wake(): void {
    if (!this.attached || !this.pending) return;
    this.attempts = 0;
    if (this.timer !== null) clearTimeout(this.timer);
    this.timer = null;
    this.run();
  }
  /** Offline and hidden views wait for a recovery signal rather than spinning a timer. */
  private schedule(): void {
    if (!this.attached || !this.pending || this.running || this.timer !== null ||
      this.attempts >= this.delays.length || !this.available()) return;
    this.timer = setTimeout(() => {this.timer = null; this.run();}, this.delays[this.attempts]);
  }
  /** Every asynchronous attempt rechecks view availability and carries a retirement token. */
  private run(): void {
    if (!this.attached || !this.pending || this.running || !this.available()) return;
    this.attempts++;
    /** This attempt alone may settle its running marker. */
    const token = {};
    /** Captured retirement revision guards asynchronous completion. */
    const revision = this.revision;
    this.running = token;
    void Promise.resolve().then(() => {
      if (this.running !== token || this.revision !== revision || !this.attached || !this.available()) return;
      return this.retry();
    }).catch(() => undefined).finally(() => {
      if (this.running !== token || this.revision !== revision) return;
      this.running = null;
      this.schedule();
    });
  }
}

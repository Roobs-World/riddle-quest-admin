import {updateLiveRace} from "../firebase";
import {createRequestId} from "./requestIds";

/** Streams private wake-up revisions through the existing authenticated callable SDK. */
export function observeFirebaseLiveRace(ticketId: string, onChange: () => void,
  onStatus: (connected: boolean) => void): () => void {
  /** Retirement cancels the network connection, reconnect timer and delayed callbacks. */
  const controller = new AbortController();
  /** Last accepted revision, reconnect backoff and its pending renewal. */
  let revision: string | undefined, failures = 0, timer: ReturnType<typeof setTimeout> | undefined;
  /** Heartbeat deadline returns silent connections to fast reconciliation. */
  let stale: ReturnType<typeof setTimeout> | undefined;
  /** A bounded stream renews credentials regularly without losing authoritative reconciliation. */
  const connect = async () => {
    /** Bounds only final response completion in this diagnostic probe. */
    let finalTimeout: ReturnType<typeof setTimeout> | undefined;
    try {
      /** Existing Firebase SDK preserves authentication and App Check on each stream. */
      const result = await updateLiveRace.stream({ticketId, action: "poll", requestId: createRequestId("live-observe")},
        {signal: controller.signal});
      // A rejected final response must always be observed, even if reading chunks fails first.
      void result.data.catch(() => {});
      for await (const chunk of result.stream) {
        if (controller.signal.aborted) return;
        /** Untrusted chunks must match the private revision contract. */
        const next = (chunk as {
          /** Opaque server revision; null keeps legacy seats on fallback polling. */
          revision?: unknown;
        } | null)?.revision;
        /** Only valid server notifications enable the slower reconciliation watchdog. */
        const connected = typeof next === "string" && /^[a-f0-9-]{36}$/.test(next);
        onStatus(connected);
        clearTimeout(stale);
        stale = setTimeout(() => {if (!controller.signal.aborted) onStatus(false);}, 15_000);
        if (connected) {
          failures = 0;
          if (revision !== next) {revision = next; onChange();}
        }
      }
      await Promise.race([result.data, new Promise<never>((_resolve, reject) => {
        finalTimeout = setTimeout(() => reject(new Error("Live observation final response timed out")), 15_000);
      })]);
      if (!controller.signal.aborted) timer = setTimeout(() => void connect(), 50);
    } catch {
      if (controller.signal.aborted) return;
      clearTimeout(stale); onStatus(false);
      timer = setTimeout(() => void connect(), Math.min(8000, 500 * 2 ** Math.min(failures++, 4)));
    } finally {clearTimeout(finalTimeout);}
  };
  void connect();
  return () => {controller.abort(); clearTimeout(timer); clearTimeout(stale);};
}

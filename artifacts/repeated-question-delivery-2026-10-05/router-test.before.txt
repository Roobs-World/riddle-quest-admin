import {describe, expect, it, jest} from "@jest/globals";
import type {GameSessionState} from "../gameSessionTypes";

/** Provides mock use memo behavior. */
const mockUseMemo = (factory: () => unknown) => factory();

jest.mock("react", () => ({
  useMemo: mockUseMemo,
  // This harness models committed renders; suspended rendering is verified in the browser.
  useRef: (value: unknown) => ({current: value}),
  useLayoutEffect: (effect: () => void) => effect(),
  useSyncExternalStore: jest.fn(),
}));

import {useGameMessageRouter} from "../useGameMessageRouter";
import {setLocaleAndWait, type Locale} from "../i18n";

describe("useGameMessageRouter", () => {
  it.each<[Locale, string]>([
    ["en", "No more letters can be revealed."],
    ["fr", "Plus aucune lettre ne peut être révélée."],
  ])("explains an exhausted Hint in %s without changing coins or awarding a Hint", async (locale, notice) => {
    /** Records only game actions generated by the genuine null response contract. */
    const dispatch = jest.fn();
    /** Default router receives the correlated success after paid-journal settlement. */
    const router = useGameMessageRouter({
      dispatch,
      stateRef: {current: {} as GameSessionState},
      postToHost: jest.fn(),
      requestRiddle: jest.fn(),
    });
    try {
      await setLocaleAndWait(locale);
      await router({type: "host:operationResult", requestId: "exhausted-hint", payload: {
        operation: "hint", ok: true, data: null,
      }});
      expect(dispatch.mock.calls).toEqual([
        [{type: "rejectHint"}],
        [{type: "setNotice", notice}],
      ]);
    } finally {
      await setLocaleAndWait("en");
    }
  });

  it.each(["profile", "leaderboard"])("retires %s presentation during its additional asynchronous validation", async (operation) => {
    /** Actual reducer-facing work must not resume after the listener's ownership ends. */
    const dispatch = jest.fn();
    /** Warm core parsers ensure this handler starts before its schema import yields. */
    const router = useGameMessageRouter({dispatch, stateRef: {current: {} as GameSessionState}, postToHost: jest.fn(), requestRiddle: jest.fn()});
    await router({type: "host:pause", payload: {}});
    dispatch.mockClear();
    /** Cleanup can occur after handler entry but before imported schema delivery. */
    let owned = true;
    /** Both operations validate a successful response, not a simulated rejection. */
    const data = operation === "profile" ? {profile: {username: "Alice", coins: 300, level: 1, xp: 0,
      winStreak: 0, gamesPlayed: 0, wins: 0, achievements: [], classicIndex: 0}} : {type: "xp", entries: []};
    /** Pending optional presentation remains bound to its originating listener. */
    const pending = router({type: "host:operationResult", payload: {operation, ok: true, data}}, () => owned);
    owned = false;
    await pending;
    expect(dispatch).not.toHaveBeenCalled();
  });

  it("does not leak a late multiplayer failure into the solo riddle notice", async () => {
    /** Provides dispatch behavior. */
    const dispatch = jest.fn();
    /** Provides router behavior. */
    const router = useGameMessageRouter({
      dispatch,
      stateRef: {current: {} as GameSessionState},
      postToHost: jest.fn(),
      requestRiddle: jest.fn(),
    });

    await router({
      type: "host:operationResult",
      requestId: "race-guess-after-close",
      payload: {
        operation: "raceGuess",
        ok: false,
        error: "This Race to 10 run is no longer active. [400]",
        errorKind: "rejected",
      },
    });

    expect(dispatch).not.toHaveBeenCalled();
  });

  it("consumes a successful bridge cancellation without notice, balance, or fanfare actions", async () => {
    /** Provides dispatch behavior. */
    const dispatch = jest.fn();
    /** Provides router behavior. */
    const router = useGameMessageRouter({
      dispatch,
      stateRef: {current: {} as GameSessionState},
      postToHost: jest.fn(),
      requestRiddle: jest.fn(),
    });

    await router({
      type: "host:operationResult",
      requestId: "purchase-cancelled-1",
      payload: {
        operation: "purchase",
        ok: true,
        data: {
          success: false,
          cancelled: true,
          productId: "coin-pack-1",
          purchaseType: "consumable",
        },
      },
    });

    expect(dispatch).not.toHaveBeenCalled();
  });
});

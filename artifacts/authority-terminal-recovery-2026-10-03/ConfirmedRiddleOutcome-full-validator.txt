import {BridgeRiddleSchema, type BridgeRiddle} from "@qwordly/core/bridge";
import type {GameSessionState} from "./gameSessionTypes";

/** Recovers a terminal win from complete host-accepted feedback, independently of provider saves. */
export const getConfirmedRiddleWin = (riddle: BridgeRiddle): GameSessionState["lastOutcome"] => {
  /** Revalidate authority history so malformed direct callers cannot manufacture completion intent. */
  const parsed = BridgeRiddleSchema.safeParse(riddle);
  if (!parsed.success) return null;
  /** The strict bridge enforces complete history, ordered attempts and no guesses after an earlier win. */
  const finalRow = parsed.data.acceptedGuesses?.at(-1);
  if (!finalRow || !finalRow.feedback.every(cell => cell.status === "green")) return null;
  return {
    id: `${riddle.sessionToken}-win-${riddle.attemptsUsed}`,
    kind: "win",
    riddleId: riddle.riddleId,
    gameMode: riddle.gameMode,
    sessionToken: riddle.sessionToken,
    attempts: riddle.attemptsUsed,
    isCorrect: true,
  };
};

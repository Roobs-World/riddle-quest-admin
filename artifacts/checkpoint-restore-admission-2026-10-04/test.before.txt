import {describe, expect, it, jest} from "@jest/globals";
import {GameSessionCheckpoints} from "../GameSessionCheckpoints";
import {createInitialGameSession, gameSessionReducer} from "../gameSession";
import {BridgeRiddleSchema, type BridgeRiddle} from "@qwordly/core/bridge";
import {computeFeedback} from "@qwordly/core/game";
import {createSavedGame} from "../GameSessionCheckpoints";
import {getConfirmedRiddleWin} from "../ConfirmedRiddleOutcome";

/** Authority-issued question without provider or answer data in its durable checkpoint. */
const riddle = {
  riddleId: "classic-checkpoint-1", sessionToken: "00000000-0000-4000-8000-000000000001",
  text: "Clue", answerLength: 5, gameMode: "classic" as const,
  attemptsUsed: 0, hasUsedContinue: false, awaitingContinue: false,
};

/** Reducer-driven active session used by callers of the checkpoint interface. */
const activeSession = () => gameSessionReducer(createInitialGameSession(), {type: "loadRiddle", riddle});

describe("GameSessionCheckpoints", () => {
  it("keeps retained drafts isolated from adapter mutation and authoritative hint updates", () => {
    /** Fresh checkpoint owner for this recovery scenario. */
    const checkpoints = new GameSessionCheckpoints();
    /** Reducer state advanced through loading, editing and restoration. */
    let state = activeSession();
    checkpoints.update(state);
    state = gameSessionReducer(state, {type: "typeLetter", letter: "X"});
    state = gameSessionReducer(state, {type: "typeLetter", letter: "I"});
    /** Adapter-facing save mutated to verify retained drafts stay isolated. */
    const saved = checkpoints.update(state);
    if (saved?.kind !== "save" || !saved.game) throw new Error("Expected draft save.");
    saved.game.currentLetters[1] = "Z";
    state = gameSessionReducer(state, {type: "loadRiddle", riddle: {
      ...riddle, revealedHints: [{index: 0, letter: "P"}],
    }});
    /** Restoration decision for the reissued authority session. */
    const restore = checkpoints.update(state);
    if (restore?.kind !== "restore") throw new Error("Expected retained draft.");
    state = gameSessionReducer(state, restore.action);
    expect(state.currentLetters).toEqual(["P", "I", "", "", ""]);
    expect(state.hintRevealedIndices).toEqual([0]);
  });

  it("discards retained drafts when the account generation changes", () => {
    /** Fresh checkpoint owner for this recovery scenario. */
    const checkpoints = new GameSessionCheckpoints();
    /** Reducer state advanced through loading, editing and restoration. */
    let state = {...activeSession(), user: {uid: "player", isAnonymous: true, accountGeneration: 1}};
    checkpoints.update(state);
    state = {...gameSessionReducer(state, {type: "typeLetter", letter: "P"}), user: state.user};
    checkpoints.update(state);
    /** New account generation that cannot inherit the previous draft. */
    const replacement = {...activeSession(), user: {...state.user, accountGeneration: 2}};
    expect(checkpoints.update(replacement)).toBeNull();
  });

  it("retains a draft without an initial provider checkpoint across another session", () => {
    /** Fresh checkpoint owner for this recovery scenario. */
    const checkpoints = new GameSessionCheckpoints();
    /** Reducer state advanced through loading, editing and restoration. */
    let state = activeSession();
    checkpoints.update(state);
    state = gameSessionReducer(state, {type: "typeLetter", letter: "P"});
    checkpoints.update(state);
    state = gameSessionReducer(state, {type: "loadRiddle", riddle: {
      ...riddle, sessionToken: "00000000-0000-4000-8000-000000000002",
    }});
    checkpoints.update(state);
    state = gameSessionReducer(state, {type: "loadRiddle", riddle: {...riddle}});
    /** Restoration decision for the reissued authority session. */
    const restore = checkpoints.update(state);
    if (restore?.kind !== "restore") throw new Error("Expected retained draft.");
    state = gameSessionReducer(state, restore.action);
    expect(state.currentLetters).toEqual(["P", "", "", "", ""]);
    expect(checkpoints.update(state)).toMatchObject({kind: "save", game: {currentLetters: ["P", "", "", "", ""]}});
  });

  it("restores the latest draft when returning from another issued locale session and saves it for reload", () => {
    /** Fresh checkpoint owner for this recovery scenario. */
    const checkpoints = new GameSessionCheckpoints();
    /** Reducer state advanced through loading, editing and restoration. */
    let state = activeSession();
    checkpoints.update(state);
    state = gameSessionReducer(state, {type: "typeLetter", letter: "P"});
    /** First persisted draft used to test later in-memory edits. */
    const initial = checkpoints.update(state);
    if (initial?.kind !== "save" || !initial.game) throw new Error("Expected initial draft save.");
    state = gameSessionReducer(state, {type: "typeLetter", letter: "I"});
    checkpoints.update(state, initial.game);
    state = gameSessionReducer(state, {type: "loadRiddle", riddle: {
      ...riddle, sessionToken: "00000000-0000-4000-8000-000000000002", text: "Autre question",
    }});
    checkpoints.update(state, initial.game);
    state = gameSessionReducer(state, {type: "typeLetter", letter: "Z"});
    checkpoints.update(state, initial.game);
    state = gameSessionReducer(state, {type: "loadRiddle", riddle: {...riddle}});
    /** Restoration decision for the reissued authority session. */
    const restore = checkpoints.update(state, initial.game);
    if (restore?.kind !== "restore") throw new Error("Expected latest draft restoration.");
    state = gameSessionReducer(state, restore.action);
    expect(state.currentLetters).toEqual(["P", "I", "", "", ""]);
    /** Save decision after restoring the latest retained draft. */
    const persisted = checkpoints.update(state, initial.game);
    expect(persisted).toMatchObject({kind: "save", game: {currentLetters: ["P", "I", "", "", ""]}});
  });

  it("keeps bounded winning proof consistent with the bridge's recovery contract", () => {
    for(const attemptsUsed of [0, 1, 4, 5, 6, 7])for(const hasUsedContinue of [false, true])
      for(const awaitingContinue of [false, true])for(const answerLength of [0, 4, 5, 7, 8])
        for(const proof of ["missing", "wrong", "win-last", "past-win", "wide-letter"]) {
          /** A bounded cross-product varies the actual wire contract's attempt and feedback evidence. */
          const candidate: BridgeRiddle = {...riddle, attemptsUsed, hasUsedContinue, awaitingContinue, answerLength,
            ...(proof === "missing" ? {} : {acceptedGuesses: Array.from({length: attemptsUsed}, (_, index) => ({attempt: index + 1,
              feedback: Array.from({length: answerLength}, () => ({letter: proof === "wide-letter" ? "AB" : "P",
                status: proof === "wrong" ? "red" : proof === "past-win" ? index === 0 ? "green" : "red" :
                  index === attemptsUsed - 1 ? "green" : "red"}))}))})};
          /** The full parser remains the authoritative oracle outside the startup bundle. */
          const parsed = BridgeRiddleSchema.safeParse(candidate);
          /** Only a valid complete history ending in green can establish terminal recovery. */
          const expected = parsed.success && Boolean(parsed.data.acceptedGuesses?.at(-1)?.feedback.every(cell => cell.status === "green"));
          expect(Boolean(getConfirmedRiddleWin(candidate))).toBe(expected);
        }
  });

  it.each(["missing", "empty", "wrong", "width", "order", "past-win", "illegal-sixth"])(
    "cannot create a terminal outcome from incomplete or invalid host recovery: %s", invalid => {
      /** Direct callers must satisfy the same canonical evidence contract as parsed bridge messages. */
      const issued: BridgeRiddle = {...riddle, attemptsUsed: 1, acceptedGuesses: [{attempt: 1,
        feedback: [..."PIANO"].map(letter => ({letter, status: "green"}))}]};
      if(invalid === "missing") delete issued.acceptedGuesses;
      if(invalid === "empty") issued.acceptedGuesses = [];
      if(invalid === "wrong") issued.acceptedGuesses![0].feedback[0].status = "red";
      if(invalid === "width") issued.acceptedGuesses![0].feedback.pop();
      if(invalid === "order") issued.acceptedGuesses![0].attempt = 2;
      if(invalid === "past-win") {
        issued.attemptsUsed = 2;
        issued.acceptedGuesses!.push({attempt: 2, feedback: [..."QQQQQ"].map(letter => ({letter, status: "red"}))});
      }
      if(invalid === "illegal-sixth") {
        issued.attemptsUsed = 6;
        issued.acceptedGuesses = Array.from({length: 6}, (_, index) => ({attempt: index + 1,
          feedback: [..."PIANO"].map(letter => ({letter, status: index === 5 ? "green" : "red"}))}));
      }
      /** Canonical proof is evaluated at the actual loaded-session seam. */
      const state = gameSessionReducer(createInitialGameSession(), {type: "loadRiddle", riddle: issued});
      expect(state.lastOutcome).toBeNull();
      expect(state.engineSnapshot?.gameOver).toBe(false);
      expect(state.modal).not.toBe("victory");
  });

  it.each([1, 5, 6])("recovers a host-accepted win on attempt %s without a current provider checkpoint", attempts => {
    /** Actual host feedback contains all accepted attempts, including the terminal winning answer. */
    const acceptedGuesses = Array.from({length: attempts}, (_, index) => {
      /** Final accepted row wins; earlier rows retain their ordinary wrong-answer feedback. */
      const guess = index === attempts - 1 ? "PIANO" : "QQQQQ";
      return {attempt: index + 1, feedback: [...guess].map((letter, cell) => ({letter,
        status: computeFeedback(guess, "PIANO")[cell]}))};
    });
    /** Strict bridge validation is the same host boundary traversed by the mounted game. */
    const issued = BridgeRiddleSchema.parse({...riddle, attemptsUsed: attempts, hasUsedContinue: attempts === 6, acceptedGuesses});
    for(const provider of [undefined, createSavedGame(gameSessionReducer(activeSession(), {type: "typeLetter", letter: "X"}))]) {
      /** Replacement page must not need the earlier draft to identify a confirmed terminal win. */
      let state = gameSessionReducer(createInitialGameSession(), {type: "loadRiddle", riddle: issued});
      /** Real checkpoint owner applies restoration before another durable save. */
      const owner = new GameSessionCheckpoints();
      /** Startup may contain no provider save or only the pre-acceptance draft. */
      const effect = owner.update(state, provider);
      expect(effect).toMatchObject({kind: "save", game: {pendingOutcome: {kind: "win", attempts}}});
      if(effect?.kind === "restore") state = gameSessionReducer(state, effect.action);
      expect(state.modal).toBe("victory");
      expect(state.engineSnapshot).toMatchObject({attemptsUsed: attempts, gameOver: true});
      expect(state.lastOutcome).toMatchObject({id: `${issued.sessionToken}-win-${attempts}`,
        kind: "win", isCorrect: true, attempts, sessionToken: issued.sessionToken});
      expect(gameSessionReducer(state, {type: "typeLetter", letter: "X"})).toBe(state);
      expect(createSavedGame(state)?.pendingOutcome).toEqual(state.lastOutcome);
      expect(createSavedGame(gameSessionReducer(state, {type: "completionAcknowledged"}))).toBeNull();
    }
  });

  it.each(["restoreQueuedGuesses", "restoreConfirmedRiddle"] as const)("cannot downgrade a host-confirmed win with an older %s", type => {
    /** The same pre-acceptance gameplay fields retained by the pending-play journal. */
    const prior = activeSession();
    /** Journal rollback base predates the already accepted winning receipt. */
    const base = {currentLetters: prior.currentLetters, activeSlotIndex: prior.activeSlotIndex,
      guesses: prior.guesses, engineSnapshot: prior.engineSnapshot, modal: prior.modal,
      lastOutcome: prior.lastOutcome, completionAcknowledged: prior.completionAcknowledged,
      letterStates: prior.letterStates, profile: prior.profile, lastRewards: prior.lastRewards,
      optimisticAchievementIds: prior.optimisticAchievementIds};
    /** Canonical winning history arrives through the ordinary host-loaded riddle. */
    const issued = {...riddle, attemptsUsed: 1, acceptedGuesses: [{attempt: 1,
      feedback: [..."PIANO"].map(letter => ({letter, status: "green" as const}))}]};
    /** Loaded winner must remain protected while older client journal actions settle. */
    const state = gameSessionReducer(createInitialGameSession(), {type: "loadRiddle", riddle: issued});
    /** Both real journal restoration actions previously copied the old nonterminal base. */
    const result = gameSessionReducer(state, type === "restoreQueuedGuesses" ?
      {type, base, guesses: [], draft: Array(5).fill("X")} : {type, base, riddle});
    expect(result.lastOutcome).toEqual(state.lastOutcome);
    expect(result.modal).toBe("victory");
    expect(result.engineSnapshot?.gameOver).toBe(true);
    expect(result.engineSnapshot?.attemptsUsed).toBe(1);
  });

  it("avoids serializing unchanged progress across presentation and profile updates", () => {
    /** Reducer snapshots retain immutable gameplay fields during unrelated updates. */
    let state = activeSession();
    /** Checkpoint owner has already observed the authority-issued question. */
    const checkpoints = new GameSessionCheckpoints();
    checkpoints.update(state);
    /** Observes real serialization without changing its result. */
    const serialize = jest.spyOn(JSON, "stringify");
    try {
      for (let index = 0; index < 100; index++) {
        state = gameSessionReducer(state, {type: "setNotice", notice: String(index)});
        expect(checkpoints.update(state)).toBeNull();
        state = gameSessionReducer(state, {type: "openModal", modal: "settings"});
        expect(checkpoints.update(state)).toBeNull();
        state = gameSessionReducer(state, {type: "closeModal"});
        expect(checkpoints.update(state)).toBeNull();
        state = gameSessionReducer(state, {type: "pause"});
        expect(checkpoints.update(state)).toBeNull();
        state = gameSessionReducer(state, {type: "resume"});
        expect(checkpoints.update(state)).toBeNull();
        state = gameSessionReducer(state, {type: "profileUpdated", profile: {
          username: "Guest", coins: index, level: 1, xp: 0, winStreak: 0,
          gamesPlayed: 0, wins: 0, achievements: [], classicIndex: 1,
        }});
        expect(checkpoints.update(state)).toBeNull();
      }
      expect(serialize).not.toHaveBeenCalled();
      state = gameSessionReducer(state, {type: "typeLetter", letter: "P"});
      expect(checkpoints.update(state)).toMatchObject({kind: "save", game: {currentLetters: ["P", "", "", "", ""]}});
      expect(serialize).toHaveBeenCalledTimes(1);
    } finally {serialize.mockRestore();}
  });

  it("forces fresh background snapshots even when an adapter mutates its earlier copy", () => {
    /** Checkpoint owner observes one immutable draft. */
    const checkpoints = new GameSessionCheckpoints();
    /** Reducer-owned draft remains separate from both returned copies. */
    const state = gameSessionReducer(activeSession(), {type: "typeLetter", letter: "P"});
    /** First lifecycle checkpoint belongs to the receiving adapter. */
    const first = checkpoints.update(state, undefined, true);
    if (first?.kind !== "save" || !first.game) throw new Error("Expected a background checkpoint.");
    first.game.currentLetters[0] = "X";
    expect(checkpoints.update({...state, paused: true})).toBeNull();
    /** Repeated lifecycle flush must project the reducer's original progress. */
    const second = checkpoints.update(state, undefined, true);
    expect(second).toMatchObject({kind: "save", game: {currentLetters: ["P", "", "", "", ""]}});
    if (second?.kind !== "save" || !second.game) throw new Error("Expected a fresh background checkpoint.");
    expect(second.game).not.toBe(first.game);
    expect(second.game.currentLetters).not.toBe(first.game.currentLetters);
  });

  it("checkpoints hint acknowledgement even when the presentation is unchanged", () => {
    /** Confirmed hint state is the only durable difference after a preview. */
    const checkpoints = new GameSessionCheckpoints();
    /** Draft from the authority-issued question. */
    let state = activeSession();
    checkpoints.update(state);
    state = gameSessionReducer(state, {type: "previewHint", index: 0, letter: "P"});
    expect(checkpoints.update(state)).toBeNull();
    state = gameSessionReducer(state, {type: "applyHint", index: 0, letter: "P"});
    expect(checkpoints.update(state)).toMatchObject({kind: "save", game: {
      currentLetters: ["P", "", "", "", ""], hintRevealedIndices: [0],
    }});
  });

  it("restores only a matching issued session, before saving, and ignores save echoes", () => {
    /** Session module owns ordered restore and save decisions. */
    const checkpoints = new GameSessionCheckpoints();
    /** Player draft assembled through reducer commands. */
    const typed = gameSessionReducer(activeSession(), {type: "typeLetter", letter: "P"});
    /** Durable checkpoint returned through the module interface. */
    const saved = new GameSessionCheckpoints().update(typed, undefined, true);
    if (saved?.kind !== "save" || !saved.game) throw new Error("Expected an active checkpoint.");

    /** Freshly issued question before checkpoint restoration. */
    const active = activeSession();
    expect(new GameSessionCheckpoints().update(active, {...saved.game, sessionToken: "different-session"})).toBeNull();
    /** First matching checkpoint must produce a restore decision. */
    const restore = checkpoints.update(active, saved.game);
    if (restore?.kind !== "restore") throw new Error("Expected restoration before a save.");
    /** Restored state remains editable despite echoed target saves. */
    let restored = gameSessionReducer(active, restore.action);
    expect(restored.currentLetters[0]).toBe("P");
    checkpoints.update(restored, saved.game);
    restored = gameSessionReducer(restored, {type: "typeLetter", letter: "I"});
    expect(checkpoints.update(restored, saved.game)).toMatchObject({kind: "save", game: {currentLetters: ["P", "I", "", "", ""]}});
    expect(checkpoints.update(restored, saved.game)).toBeNull();
  });

  it("never treats a delayed echo of its first save as startup restoration", () => {
    /** Session module consumes startup restoration when the authority question first arrives. */
    const checkpoints = new GameSessionCheckpoints();
    /** Player starts without a stored checkpoint. */
    let state = activeSession();
    checkpoints.update(state);
    state = gameSessionReducer(state, {type: "typeLetter", letter: "P"});
    /** First durable save can be echoed after more input arrives. */
    const saved = checkpoints.update(state);
    if (saved?.kind !== "save") throw new Error("Expected the first draft checkpoint.");
    state = gameSessionReducer(state, {type: "typeLetter", letter: "I"});
    expect(checkpoints.update(state, saved.game)).toMatchObject({kind: "save", game: {currentLetters: ["P", "I", "", "", ""]}});
  });

  it("checkpoints authoritative feedback corrections even when the row count is unchanged", () => {
    /** Session module owns ordered restore and save decisions. */
    const checkpoints = new GameSessionCheckpoints();
    /** Reducer-owned gameplay supplied through the checkpoint interface. */
    let state = gameSessionReducer(activeSession(), {type: "guessResolved",
      feedback: {guess: [..."world"], feedback: ["red", "red", "red", "red", "red"]},
      attempts: 1, isCorrect: false, gameOver: false, canContinue: false});
    checkpoints.update(state);
    state = {...state, guesses: [{guess: [..."world"], feedback: ["yellow", "red", "red", "red", "red"]}]};
    expect(checkpoints.update(state)).toMatchObject({kind: "save", game: {
      guesses: [{guess: [..."world"], feedback: ["yellow", "red", "red", "red", "red"]}],
    }});
  });

  it("keeps a provisional Continue out of durable progress until its authority acknowledgement", () => {
    /** Session module owns ordered restore and save decisions. */
    const checkpoints = new GameSessionCheckpoints();
    /** Reducer-owned gameplay supplied through the checkpoint interface. */
    let state = gameSessionReducer(createInitialGameSession(), {type: "hostInit", locale: "en", gameMode: "classic",
      user: null, products: [], profile: {username: "Guest", coins: 200, level: 1, xp: 0,
        winStreak: 0, gamesPlayed: 0, wins: 0, achievements: [], classicIndex: 1}});
    state = gameSessionReducer(state, {type: "loadRiddle", riddle: {...riddle, attemptsUsed: 5, awaitingContinue: true}});
    state = gameSessionReducer(state, {type: "previewCoinSpend", requestId: "continue-1", amount: 100, reason: "continue"});
    expect(state.engineSnapshot?.hasUsedContinue).toBe(true);
    expect(checkpoints.update(state, undefined, true)).toMatchObject({kind: "save", game: {
      awaitingContinue: true, engineSnapshot: {attemptsUsed: 5, hasUsedContinue: false},
    }});
    state = gameSessionReducer(state, {type: "confirmCoinSpend", requestId: "continue-1", newCoins: 100});
    expect(checkpoints.update(state)).toMatchObject({kind: "save", game: {
      awaitingContinue: false, engineSnapshot: {attemptsUsed: 5, hasUsedContinue: true},
    }});
  });

  it("excludes unconfirmed hints and preserves the input replaced by their preview", () => {
    /** Reducer-owned gameplay supplied through the checkpoint interface. */
    let state = gameSessionReducer(activeSession(), {type: "typeLetter", letter: "X"});
    state = gameSessionReducer(state, {type: "previewHint", index: 0, letter: "P"});
    expect(state.currentLetters[0]).toBe("P");
    /** Durable checkpoint returned through the module interface. */
    const saved = new GameSessionCheckpoints().update(state, undefined, true);
    expect(saved).toMatchObject({kind: "save", game: {currentLetters: ["X", "", "", "", ""], hintRevealedIndices: []}});
  });

  it("isolates adapter-owned snapshots from session state and clears acknowledged outcomes once", () => {
    /** Session module owns ordered restore and save decisions. */
    const checkpoints = new GameSessionCheckpoints();
    /** Reducer-owned gameplay supplied through the checkpoint interface. */
    let state = gameSessionReducer(activeSession(), {type: "guessResolved",
      feedback: {guess: [..."piano"], feedback: ["green", "green", "green", "green", "green"]},
      attempts: 1, isCorrect: true, gameOver: true, canContinue: false});
    /** Durable checkpoint returned through the module interface. */
    const saved = checkpoints.update(state, undefined, true);
    if (saved?.kind !== "save" || !saved.game) throw new Error("Expected a pending completion checkpoint.");
    saved.game.currentLetters[0] = "X";
    saved.game.guesses[0].guess[0] = "X";
    saved.game.engineSnapshot.guesses[0].feedback[0] = "red";
    if (saved.game.pendingOutcome) saved.game.pendingOutcome.attempts = 99;
    expect(state.guesses[0].guess[0]).toBe("p");
    expect(state.engineSnapshot?.guesses[0].feedback[0]).toBe("green");
    expect(state.lastOutcome?.attempts).toBe(1);

    state = gameSessionReducer(state, {type: "completionAcknowledged"});
    expect(checkpoints.update(state)).toEqual({kind: "save", game: null});
    expect(checkpoints.update(state, undefined, true)).toBeNull();
  });
});

import {afterAll,beforeAll,describe,it,jest,expect as assert} from "@jest/globals";
import {chromium,expect,type Browser} from "@playwright/test";
import {createServer,type Server} from "node:http";
import {readFile,writeFile} from "node:fs/promises";
import path from "node:path";
import {openAuthenticationHarness} from "/home/craig-beswetherick/Documents/riddle-quest/packages/portal/src/jest/__tests__/authenticationHarness.jest-playwright.support";
import {enterRegisteredGame} from "/home/craig-beswetherick/Documents/riddle-quest/packages/portal/src/jest/__tests__/authentication.jest-playwright.support";
jest.setTimeout(60000);
const repo="/home/craig-beswetherick/Documents/riddle-quest";
const artifact=path.join(repo,"artifacts/jest-e2e-2026-10-22/jest-artifact-journal");
const out=path.join(repo,"artifacts/jest-bll-2026-10-23/purchase-completion");
let browser:Browser,server:Server,origin:string;
beforeAll(async()=>{
 server=createServer(async(req,res)=>{
  const requestPath=new URL(req.url??"/","http://127.0.0.1").pathname;
  const filename=path.join(artifact,requestPath==="/"?"index.html":requestPath);
  try {const body=await readFile(path.extname(filename)?filename:path.join(artifact,"index.html"));
   const mime:Record<string,string>={".js":"application/javascript",".css":"text/css",".html":"text/html",".json":"application/json",".svg":"image/svg+xml",".webp":"image/webp",".png":"image/png",".woff2":"font/woff2",".ttf":"font/ttf"};
   res.writeHead(200,{"Content-Type":mime[path.extname(filename)]??"application/octet-stream"});res.end(body);
  }catch{res.writeHead(404);res.end();}
 });
 await new Promise<void>(resolve=>server.listen(0,"127.0.0.1",resolve));
 const address=server.address();if(!address||typeof address==="string")throw Error("No fixture port");
 origin=`http://127.0.0.1:${address.port}`;
 browser=await chromium.launch({headless:true});
});
afterAll(async()=>{await browser?.close();await new Promise<void>(resolve=>server?.close(()=>resolve()));});
describe("actual compiled Jest purchase presentation",()=>{
 it.each(["success","returned-failure","thrown-failure"])("displays the confirmed grant after provider completion mode=%s",async(failCompletion)=>{
  const {page,context,pageErrors}=await openAuthenticationHarness(browser,origin,{registered:true,scenario:"localized-coin-catalog"});
  try{
   await enterRegisteredGame(page);
   const balance=page.locator("[data-coin-meter] strong");
   await expect(balance).toHaveText("300");
   await page.evaluate(fail=>{
    const sdk=Reflect.get(window,"JestSDK");
    const proof={completionAttempts:0,beginAttempts:0,pending:false};Reflect.set(window,"__BLL_PURCHASE_PROOF__",proof);
    sdk.payments.beginPurchase=async({productSku}:{productSku:string})=>{
     proof.beginAttempts+=1;proof.pending=true;
     return {result:"success",purchase:{purchaseToken:"synthetic-bll-proof",productSku,sandbox:true},purchaseSigned:"synthetic-bll-envelope"};
    };
    sdk.payments.completePurchase=async()=>{
     proof.completionAttempts+=1;
     if(fail==="thrown-failure")throw Error("mock_completion_offline");
     if(fail==="returned-failure")return {result:"error",error:"mock_completion_offline"};
     proof.pending=false;return {result:"success"};
    };
   },failCompletion);
   await page.getByRole("button",{name:/buy coins/i}).first().click();
   const offer=page.locator(".game-shop-grid article").filter({hasText:"Provider custom description"});
   await offer.getByRole("button").click();
   await page.waitForFunction(()=>Reflect.get(window,"__BLL_PURCHASE_PROOF__").completionAttempts===1);
   await page.waitForTimeout(250);
   const observed={failCompletion,balance:await balance.innerText(),...(await page.evaluate(()=>Reflect.get(window,"__BLL_PURCHASE_PROOF__"))),pageErrors};
   await page.screenshot({path:path.join(out,`completion-${failCompletion}.png`)});
   await writeFile(path.join(out,`completion-${failCompletion}.json`),JSON.stringify(observed,null,2));
   await expect(balance).toHaveText("550",{timeout:3000});
   assert(pageErrors).toEqual([]);
  }finally{await context.close();}
 });
});
